import numpy as np
import matplotlib.pyplot as plt
from PIL import Image
from tkinter import Tk, filedialog

# Fonction qui calcule les moyennes de blocs de pixels
def reduire_matrice(matrice, taille):

    hauteur, largeur, canaux = matrice.shape

    nouvelle = np.zeros(
        (hauteur // taille, largeur // taille, 3)
    )

    for i in range(0, hauteur, taille):
        for j in range(0, largeur, taille):

            bloc = matrice[i:i+taille, j:j+taille]

            moyenne = np.mean(bloc, axis=(0, 1))

            nouvelle[i//taille][j//taille] = moyenne

    return nouvelle
